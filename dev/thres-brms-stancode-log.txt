   real sratio_logit_lpmf(int y, real mu, real disc, vector thres) {
     int nthres = num_elements(thres);
     vector[nthres + 1] p;
     vector[nthres] q;
     int k = 1;
     while (k <= min(y, nthres)) {
       q[k] = log1m_inv_logit(disc * (thres[k] - mu));
       p[k] = log1m_exp(q[k]);
